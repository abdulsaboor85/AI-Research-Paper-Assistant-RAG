def login_required(view):
    @wraps(view)
    def wrapper(*args, **kwargs):
        if "user_id" not in session:
            return jsonify({"error": "Not logged in."}), 401
        return view(*args, **kwargs)
    return wrapper


@app.errorhandler(Exception)
def handle_error(exc):
    """Any error inside a route becomes a JSON error for the frontend."""
    if isinstance(exc, HTTPException):
        return exc
    traceback.print_exc()
    return jsonify({"error": str(exc)}), 400


def require_api_key() -> str:
    """The user's own saved key first, then GEMINI_API_KEY from .env."""
    key = get_gemini_api_key(session["user_id"]) or os.getenv("GEMINI_API_KEY")
    if not key:
        raise ValueError("No Gemini API key configured. Click the gear icon to add one.")
    return key


# =========================================================
# INDEX STATUS  (per paper collection)
# =========================================================

_index_status: dict[str, dict] = {}
_index_lock = threading.Lock()


def set_index_status(name: str, status: str, step: str = "", pct: int = 0, message: str = "") -> None:
    with _index_lock:
        _index_status[name] = {"status": status, "step": step, "pct": pct, "message": message}


def get_index_status(name: str) -> dict:
    with _index_lock:
        return _index_status.get(name, {"status": "ready", "step": "", "pct": 100, "message": ""})


# =========================================================
# CACHE  (RAM + disk). Always stores the BASE result;
# personalization is applied fresh on every request.
# =========================================================

_JSON_TYPES = {"analysis", "chat"}          # "prereqs" is plain text
_ram: dict[str, dict[str, Any]] = {"analysis": {}, "prereqs": {}, "chat": {}}
_cache_lock = threading.Lock()


def _cache_path(cache_type: str, name: str) -> Path:
    safe = re.sub(r"[^a-z0-9_]", "", name.lower())
    return CACHE_DIR / f"{safe}_{cache_type}.json"


def cache_get(cache_type: str, name: str, default: Any = None) -> Any:
    with _cache_lock:
        if name in _ram[cache_type]:
            return _ram[cache_type][name]
        path = _cache_path(cache_type, name)
        if path.exists():
            try:
                raw = path.read_text(encoding="utf-8")
                value = json.loads(raw) if cache_type in _JSON_TYPES else raw
                _ram[cache_type][name] = value
                return value
            except Exception as e:
                print(f"[cache] Could not read {cache_type} cache: {e}")
    return default


def cache_set(cache_type: str, name: str, value: Any) -> None:
    with _cache_lock:
        _ram[cache_type][name] = value
        try:
            text = json.dumps(value, ensure_ascii=False, indent=2) if cache_type in _JSON_TYPES else value
            _cache_path(cache_type, name).write_text(text, encoding="utf-8")
        except Exception as e:
            print(f"[cache] Could not write {cache_type} cache: {e}")


def cache_delete(cache_type: str, name: str) -> None:
    with _cache_lock:
        _ram[cache_type].pop(name, None)
        _cache_path(cache_type, name).unlink(missing_ok=True)


def append_chat(name: str, question: str, reply: str) -> None:
    history = list(cache_get("chat", name, []))
    history += [{"role": "user", "text": question}, {"role": "assistant", "text": reply}]
    cache_set("chat", name, history)


# =========================================================
# PAPER HELPERS
# =========================================================

TIMESTAMP_PREFIX = re.compile(r"^\d{8}_\d{6}_")


def get_user_upload_dir() -> Path:
    user_dir = UPLOAD_DIR / str(session.get("user_id", "anonymous"))
    user_dir.mkdir(parents=True, exist_ok=True)
    return user_dir


def relative_path(path: Path) -> str:
    return path.relative_to(BASE_DIR).as_posix()


def normalize_pdf_path(raw_path: str) -> Path:
    """Turn a path from the frontend into a safe PDF path inside THIS user's uploads."""
    candidate = (BASE_DIR / raw_path.strip()).resolve()
    try:
        candidate.relative_to(get_user_upload_dir().resolve())
    except ValueError as exc:
        raise ValueError("Path must stay inside your uploads folder.") from exc
    if candidate.suffix.lower() != ".pdf":
        raise ValueError("File must be a PDF.")
    if not candidate.exists():
        raise FileNotFoundError(f"PDF not found: {raw_path}")
    return candidate


def pdf_from_request() -> Path:
    data = request.get_json(silent=True) or {}
    raw_path = data.get("paper_path")
    if not raw_path:
        raise ValueError("paper_path is required.")
    return normalize_pdf_path(str(raw_path))


def paper_title(pdf_path: Path) -> str:
    return TIMESTAMP_PREFIX.sub("", pdf_path.stem)


def collection_name_from_path(pdf_path: Path) -> str:
    folder = pdf_path.parent.relative_to(BASE_DIR).as_posix().lower()
    combined = f"{folder}/{paper_title(pdf_path).lower()}"
    return f"paper_{re.sub(r'[^a-z0-9]+', '_', combined).strip('_') or 'default'}"


def paper_record(pdf_path: Path) -> dict[str, Any]:
    name = collection_name_from_path(pdf_path)
    status = get_index_status(name)
    title = paper_title(pdf_path)
    return {
        "id": relative_path(pdf_path),
        "title": title,
        "filename": title,
        "path": relative_path(pdf_path),
        "collectionName": name,
        "modifiedAt": datetime.fromtimestamp(pdf_path.stat().st_mtime).isoformat(),
        "indexStatus": status["status"],
        "indexPct": status["pct"],
        "indexStep": status["step"],
        "indexMessage": status["message"],
    }


# =========================================================
# INDEXING  (runs in a background thread)
# =========================================================

def index_paper_background(pdf_path: Path, name: str) -> None:
    set_index_status(name, "indexing", "Starting...", 5, "Preparing to index")

    def run() -> None:
        try:
            if embedder.collection_exists_and_has_data(name):
                set_index_status(name, "ready", "Done", 100, "Ready (cached)")
                return

            set_index_status(name, "indexing", "Extracting text", 15, "Reading PDF text...")
            text = extract_text(str(pdf_path))
            if len(text.strip()) < 100:
                raise ValueError("Could not extract enough text from PDF.")

            set_index_status(name, "indexing", "Chunking", 35, "Splitting into chunks...")
            chunks = chunk_text(text)
            if not chunks:
                raise ValueError("Chunking produced no chunks.")

            set_index_status(name, "indexing", "Embedding", 55, f"Embedding {len(chunks)} chunks...")
            embedder.embed_and_store(chunks, name)

            set_index_status(name, "ready", "Done", 100, "Ready")
        except Exception as exc:
            set_index_status(name, "error", "Failed", 0, str(exc))
            print(f"[indexer] {name} failed: {exc}")

    threading.Thread(target=run, daemon=True).start()


def analyze_paper(pdf_path: Path, api_key: str) -> dict[str, Any]:
    """BASE (unpersonalized) analysis. Gemini runs only once per paper."""
    name = collection_name_from_path(pdf_path)

    cached = cache_get("analysis", name)
    if cached:
        return cached

    from difficulty_scorer import analyze_difficulty  # heavy import: load only when needed

    text = extract_text(str(pdf_path))
    if len(text.strip()) < 100:
        raise ValueError("Could not extract enough text.")

    result = analyze_difficulty(full_text=text, api_key=api_key)
    result["paper"] = paper_record(pdf_path)
    cache_set("analysis", name, result)
    return result

@app.post("/api/upload")
@login_required
def api_upload():
    uploaded = request.files.get("file")
    if not uploaded or not uploaded.filename:
        return jsonify({"error": "No file uploaded."}), 400
    if not uploaded.filename.lower().endswith(".pdf"):
        return jsonify({"error": "Only PDF files allowed."}), 400

    stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    saved_path = get_user_upload_dir() / f"{stamp}_{secure_filename(uploaded.filename)}"
    uploaded.save(saved_path)

    index_paper_background(saved_path, collection_name_from_path(saved_path))
    return jsonify({"paper": paper_record(saved_path), "analysis": {}})

@app.post("/api/analyze")
@login_required
def api_analyze():
    pdf_path = pdf_from_request()
    base_result = analyze_paper(pdf_path, require_api_key())

    known_terms = get_known_terms_set(session["user_id"])
    base_prereqs = cache_get("prereqs", collection_name_from_path(pdf_path))
    result = personalize_analysis(base_result, known_terms, base_prereqs)

    paper = result.pop("paper", None) or paper_record(pdf_path)
    return jsonify({"paper": paper, "analysis": result})


@app.post("/api/prerequisites")
@login_required
def api_prerequisites():
    pdf_path = pdf_from_request()
    name = collection_name_from_path(pdf_path)

    prereqs = cache_get("prereqs", name)
    if prereqs is None:
        api_key = require_api_key()
        text = extract_text(str(pdf_path))
        if len(text.strip()) < 100:
            raise ValueError("Could not extract enough text.")
        prereqs = extract_prerequisites(text, api_key)
        cache_set("prereqs", name, prereqs)

    personalized = personalize_prerequisites(prereqs, get_known_terms_set(session["user_id"]))
    removed = len(parse_prereq_topics(prereqs)) - len(parse_prereq_topics(personalized))
    return jsonify({"prerequisites": personalized, "filtered_count": removed})

@app.post("/api/chat")
@login_required
def api_chat():
    data = request.get_json(silent=True) or {}
    question = (data.get("message") or "").strip()
    if not question:
        return jsonify({"error": "message is required."}), 400

    api_key = require_api_key()
    name = collection_name_from_path(pdf_from_request())

    status = get_index_status(name)["status"]
    if status == "indexing":
        return jsonify({"error": "Paper is still being indexed. Please wait."}), 409
    if status == "error":
        return jsonify({"error": "Indexing failed. Try re-uploading."}), 500

    chunks = retrieve_relevant_chunks(question, collection_name=name)
    reply = answer_question(question, chunks, api_key)
    append_chat(name, question, reply)

    return jsonify({"reply": reply, "collectionName": name, "chunksUsed": len(chunks)})

# auth_routes.py

"""Login / signup / logout routes. Registered in app.py with app.register_blueprint(auth_bp)."""

from pathlib import Path

from flask import Blueprint, jsonify, redirect, request, send_from_directory, session

from database import create_user, verify_user

FRONTEND_DIR = Path(__file__).resolve().parent / "frontend"

auth_bp = Blueprint("auth", __name__)


def _log_in(user: dict) -> None:
    session.clear()
    session["user_id"] = user["id"]
    session["username"] = user["username"]
    session.permanent = True


# ── Pages ────────────────────────────────────────────────

@auth_bp.get("/auth")
def auth_page():
    if "user_id" in session:
        return redirect("/")
    return send_from_directory(FRONTEND_DIR, "auth.html")


@auth_bp.get("/auth.css")
@auth_bp.get("/auth.js")
def auth_asset():
    return send_from_directory(FRONTEND_DIR, request.path.lstrip("/"))


# ── API ──────────────────────────────────────────────────

@auth_bp.post("/api/signup")
def api_signup():
    data = request.get_json(silent=True) or {}
    username = (data.get("username") or "").strip()
    email = (data.get("email") or "").strip()
    password = data.get("password") or ""

    if not (username and email and password):
        return jsonify({"error": "All fields are required."}), 400

    result = create_user(username, email, password)
    if not result["ok"]:
        return jsonify({"error": result["error"]}), 400

    _log_in(result["user"])
    return jsonify({"message": "Account created successfully!", "username": result["user"]["username"]})


@auth_bp.post("/api/login")
def api_login():
    data = request.get_json(silent=True) or {}
    identifier = (data.get("username") or data.get("email") or "").strip()
    password = data.get("password") or ""

    if not identifier or not password:
        return jsonify({"error": "Username/email and password are required."}), 400

    result = verify_user(identifier, password)
    if not result["ok"]:
        return jsonify({"error": result["error"]}), 401

    _log_in(result["user"])
    return jsonify({"message": "Login successful!", "username": result["user"]["username"]})


@auth_bp.post("/api/logout")
def api_logout():
    session.clear()
    return jsonify({"message": "Logged out successfully."})


@auth_bp.get("/api/me")
def api_me():
    if "user_id" not in session:
        return jsonify({"logged_in": False}), 401
    return jsonify({
        "logged_in": True,
        "user_id": session["user_id"],
        "username": session["username"],
    })

# frontend/auth.js

/* ==========================================================
   auth.js  —  PaperMind login / signup logic
   ========================================================== */

const API = "http://127.0.0.1:5000";

/* ── Dark mode ───────────────────────────────────────────── */

function toggleAuthDark() {
  const isDark = document.body.classList.toggle("dark");
  const btn    = document.getElementById("authDarkBtn");
  if (btn) btn.textContent = isDark ? "☀️" : "🌙";
  localStorage.setItem("papermind_darkMode", String(isDark));
}

// Apply saved dark mode on load
(function applyDark() {
  if (localStorage.getItem("papermind_darkMode") === "true") {
    document.body.classList.add("dark");
    const btn = document.getElementById("authDarkBtn");
    if (btn) btn.textContent = "☀️";
  }
})();

/* ── Mode switching ──────────────────────────────────────── */

function switchMode(mode) {
  const isLogin = mode === "login";

  document.getElementById("loginForm").style.display  = isLogin ? "" : "none";
  document.getElementById("signupForm").style.display = isLogin ? "none" : "";

  document.getElementById("loginTab").classList.toggle("active",  isLogin);
  document.getElementById("signupTab").classList.toggle("active", !isLogin);

  clearErrors();
}

function clearErrors() {
  ["loginError", "signupError"].forEach(id => {
    const el = document.getElementById(id);
    if (el) {
      el.style.display = "none";
      el.textContent   = "";
    }
  });
  document.querySelectorAll(".field-input").forEach(i => {
    i.classList.remove("error");
  });
}

/* ── Show error ──────────────────────────────────────────── */

function showError(formType, message) {
  const el = document.getElementById(formType + "Error");
  if (!el) return;
  el.textContent    = message;
  el.style.display  = "block";
}

/* ── Loading state ───────────────────────────────────────── */

function setLoading(formType, loading) {
  const btn     = document.getElementById(formType + "Btn");
  const text    = document.getElementById(formType + "BtnText");
  const spinner = document.getElementById(formType + "Spinner");

  if (!btn) return;
  btn.disabled           = loading;
  text.style.display     = loading ? "none" : "";
  spinner.style.display  = loading ? "inline-block" : "none";
}

/* ── Password visibility toggle ─────────────────────────── */

function togglePass(inputId, btn) {
  const input = document.getElementById(inputId);
  if (!input) return;

  const show = input.type === "password";
  input.type = show ? "text" : "password";

  // Eye open (visible) vs eye with slash (hidden)
  btn.innerHTML = show
    ? `<svg viewBox="0 0 20 20" fill="none" stroke="currentColor" stroke-width="1.6">
         <path d="M1 10s3.5-6 9-6 9 6 9 6-3.5 6-9 6-9-6-9-6z"/>
         <circle cx="10" cy="10" r="2.5"/>
         <line x1="3" y1="3" x2="17" y2="17"/>
       </svg>`
    : `<svg viewBox="0 0 20 20" fill="none" stroke="currentColor" stroke-width="1.6">
         <path d="M1 10s3.5-6 9-6 9 6 9 6-3.5 6-9 6-9-6-9-6z"/>
         <circle cx="10" cy="10" r="2.5"/>
       </svg>`;
}

/* ── Enter key handler ───────────────────────────────────── */

function handleKey(event, formType) {
  if (event.key === "Enter") {
    event.preventDefault();
    if (formType === "login")  doLogin();
    if (formType === "signup") doSignup();
  }
}

/* ── LOGIN ───────────────────────────────────────────────── */

async function doLogin() {
  clearErrors();

  const usernameEl = document.getElementById("loginUser");
  const passwordEl = document.getElementById("loginPass");

  const username = usernameEl.value.trim();
  const password = passwordEl.value;

  if (!username) {
    usernameEl.classList.add("error");
    showError("login", "Please enter your username or email.");
    return;
  }

  if (!password) {
    passwordEl.classList.add("error");
    showError("login", "Please enter your password.");
    return;
  }

  setLoading("login", true);

  try {
    const response = await fetch(`${API}/api/login`, {
      method:  "POST",
      headers: { "Content-Type": "application/json" },
      body:    JSON.stringify({ username, password }),
    });

    const data = await response.json();

    if (!response.ok) {
      passwordEl.classList.add("error");
      showError("login", data.error || "Login failed. Please check your credentials.");
      return;
    }

    window.location.href = "/";

  } catch {
    showError("login", "Cannot reach the server. Make sure Flask is running.");
  } finally {
    setLoading("login", false);
  }
}

/* ── SIGNUP ──────────────────────────────────────────────── */

async function doSignup() {
  clearErrors();

  const usernameEl = document.getElementById("signupUser");
  const emailEl    = document.getElementById("signupEmail");
  const passwordEl = document.getElementById("signupPass");

  const username = usernameEl.value.trim();
  const email    = emailEl.value.trim();
  const password = passwordEl.value;

  let hasError = false;

  if (!username || username.length < 3) {
    usernameEl.classList.add("error");
    hasError = true;
  }

  if (!email || !email.includes("@")) {
    emailEl.classList.add("error");
    hasError = true;
  }

  if (!password || password.length < 6) {
    passwordEl.classList.add("error");
    hasError = true;
  }

  if (hasError) {
    showError("signup", "Please fill in all fields correctly before continuing.");
    return;
  }

  setLoading("signup", true);

  try {
    const response = await fetch(`${API}/api/signup`, {
      method:  "POST",
      headers: { "Content-Type": "application/json" },
      body:    JSON.stringify({ username, email, password }),
    });

    const data = await response.json();

    if (!response.ok) {
      showError("signup", data.error || "Signup failed. Please try again.");

      if (data.error && data.error.toLowerCase().includes("username")) {
        usernameEl.classList.add("error");
      } else if (data.error && data.error.toLowerCase().includes("email")) {
        emailEl.classList.add("error");
      }

      return;
    }

    window.location.href = "/";

  } catch {
    showError("signup", "Cannot reach the server. Make sure Flask is running.");
  } finally {
    setLoading("signup", false);
  }
}

/* ── Expose to HTML onclick ──────────────────────────────── */

window.toggleAuthDark = toggleAuthDark;
window.switchMode     = switchMode;
window.doLogin        = doLogin;
window.doSignup       = doSignup;
window.togglePass     = togglePass;
window.handleKey      = handleKey;

function formatMessageText(raw) {
  if (!raw) return "";
  const bold = (s) => s.replace(/\*\*(.+?)\*\*/g, "<strong>$1</strong>");

  let html = "";
  let list = [];
  let listTag = null;
  let paragraph = [];

  const flushList = () => {
    if (list.length) html += `<${listTag}>${list.map((li) => `<li>${li}</li>`).join("")}</${listTag}>`;
    list = [];
    listTag = null;
  };
  const flushParagraph = () => {
    if (paragraph.length) html += `<p>${paragraph.join("<br>")}</p>`;
    paragraph = [];
  };
  const addListItem = (tag, text) => {
    flushParagraph();
    if (listTag !== tag) flushList();
    listTag = tag;
    list.push(bold(text));
  };

  for (const line of escapeHtml(raw.trim()).split("\n")) {
    const text = line.trim();
    if (!text) { flushList(); flushParagraph(); continue; }

    const heading = text.match(/^#{1,3}\s+(.*)/);
    const ordered = text.match(/^(\d+)[.)]\s+(.*)/);
    const bullet = text.match(/^[-*]\s+(.*)/);

    if (heading) {
      flushList(); flushParagraph();
      html += `<h4 class="msg-heading">${bold(heading[1])}</h4>`;
    } else if (ordered) {
      addListItem("ol", ordered[2]);
    } else if (bullet) {
      addListItem("ul", bullet[1]);
    } else {
      flushList();
      paragraph.push(bold(text));
    }
  }
  flushList(); flushParagraph();
  return html;
}

/** Draw one PDF page (canvas + selectable text layer) into `container`. */
async function renderPdfPage(pageNum, container, targetWidth, zoom) {
  const page = await pdfDoc.getPage(pageNum);
  const dpr = window.devicePixelRatio || 1;
  const cssScale = (targetWidth / page.getViewport({ scale: 1 }).width) * zoom;
  const viewport = page.getViewport({ scale: cssScale });
  const renderViewport = page.getViewport({ scale: cssScale * dpr });

  const wrap = document.createElement("div");
  wrap.className = "pdf-page-wrap";
  wrap.style.width = `${viewport.width}px`;
  wrap.style.height = `${viewport.height}px`;

  const canvas = document.createElement("canvas");
  canvas.width = Math.ceil(renderViewport.width);
  canvas.height = Math.ceil(renderViewport.height);
  canvas.style.cssText = `width:${viewport.width}px;height:${viewport.height}px;display:block;border-radius:6px;`;
  await page.render({ canvasContext: canvas.getContext("2d", { alpha: false }), viewport: renderViewport }).promise;
  wrap.appendChild(canvas);

  void buildPdfTextLayer(page, wrap, viewport);

  try { container.querySelector(".pdf-page-wrap")?._textLayerTask?.cancel?.(); } catch (_) {}
  clearElement(container);
  container.appendChild(wrap);
}

/* ── Select text in the PDF -> "Explain" popup ───────────── */

async function buildPdfTextLayer(page, wrap, viewport) {
  const layer = document.createElement("div");
  layer.className = "pdf-text-layer";
  layer.style.width = `${viewport.width}px`;
  layer.style.height = `${viewport.height}px`;
  layer.style.setProperty("--scale-factor", String(viewport.scale));
  wrap.appendChild(layer);

  try {
    if (typeof pdfjsLib.renderTextLayer !== "function") return;
    const textContent = await page.getTextContent({ normalizeWhitespace: true });
    // Not awaited on purpose: a slow text layer must never block the page itself.
    wrap._textLayerTask = pdfjsLib.renderTextLayer({
      textContent, container: layer, viewport, textDivs: [], enhanceTextSelection: true,
    });
    wrap._textLayerTask.promise?.catch((err) => console.warn("[PDF] text layer failed:", err));
  } catch (err) {
    console.warn("[PDF] could not build text layer:", err);
  }
}

function getPdfSelectionTerm() {
  const sel = window.getSelection();
  if (!sel || sel.rangeCount === 0 || sel.isCollapsed) return null;

  const range = sel.getRangeAt(0);
  const layerOf = (node) => (node?.nodeType === Node.ELEMENT_NODE ? node : node?.parentElement)?.closest?.(".pdf-text-layer");
  const startLayer = layerOf(range.startContainer);
  if (!startLayer || startLayer !== layerOf(range.endContainer)) return null;

  const term = sel.toString()
    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, " ")
    .replace(/[\u00A0\u200B\u200C\u200D\uFEFF]/g, " ")
    .replace(/\s+/g, " ")
    .trim();

  if (!term || term.length > 500 || term.split(" ").length > 40) return null;
  return term;
}

/** Keep the popup fully on screen (flip above the selection if needed). */
function positionPopup(popup) {
  const gap = 10, margin = 8;
  const x = Number(popup.dataset.anchorX || 0);
  const y = Number(popup.dataset.anchorY || 0);
  const rect = popup.getBoundingClientRect();
  const vw = document.documentElement.clientWidth;
  const vh = document.documentElement.clientHeight;

  let left = Math.max(margin, Math.min(x + gap, vw - rect.width - margin));
  let top = y + gap;
  if (top + rect.height > vh - margin) top = y - rect.height - gap;
  if (top < margin) top = Math.min(Math.max(margin, y + gap), vh - rect.height - margin);

  popup.style.left = `${left}px`;
  popup.style.top = `${Math.max(margin, top)}px`;
}

function startIndexPolling(paper) {
  const name = paper.collectionName;
  if (state.pollTimers[name]) return;

  const stop = () => {
    clearInterval(state.pollTimers[name]);
    delete state.pollTimers[name];
    hideProgressBar();
  };

  state.pollTimers[name] = setInterval(async () => {
    try {
      const res = await fetch(`${API_BASE}/api/status/${paper.path}`);
      if (!res.ok) return;
      const { status, pct, step, message } = await res.json();

      state.papers = state.papers.map((p) =>
        p.id !== paper.id ? p : { ...p, indexStatus: status, indexPct: pct, indexStep: step, indexMessage: message }
      );
      renderPapers();

      if (state.activePaperId === paper.id && status === "indexing") showProgressBar(pct, message || step);

      if (status === "ready") { stop(); showToast(`"${paper.title}" is ready!`); }
      if (status === "error") { stop(); showToast(`Indexing failed for "${paper.title}". Try re-uploading.`); }
    } catch { /* network blip: try again next tick */ }
  }, 2000);
}

    const item = document.createElement("div");

  clearElement(grid);

  if (clearBtn) clearBtn.style.display = n ? "inline-flex" : "none";
  grid.style.display = n ? "grid" : "none";
  empty.style.display = n ? "none" : "flex";

  state.knownTerms.forEach((term) => {
    const card = document.createElement("div");
    card.className = "memory-term-card";
    card.innerHTML = `<span class="memory-term-label"></span><button class="memory-term-remove" type="button">🗑</button>`;

    remove.addEventListener("click", (e) => { e.stopPropagation(); deleteKnownTerm(term); });

  });
}

/** "3. Softmax: explanation" -> { number, concept, explanation } */
function parsePrerequisites(text) {
  const items = [];
  text.trim().split("\n").forEach((line) => {
    const match = line.match(/^\s*(\d+)[.)]\s*(.+)/);
    if (!match) return;
    const content = match[2].trim();
    const colon = content.indexOf(":");
    items.push(colon === -1
      ? { number: match[1], concept: content, explanation: "" }
      : { number: match[1], concept: content.slice(0, colon).trim(), explanation: content.slice(colon + 1).trim() });
  });
  return items;
}

function startExplainProgress() {
  let stopped = false;

  const setProgress = (pct, label) => {
    const fill = $("explainProgressFill");
    if (fill) fill.style.width = `${pct}%`;
    setText("explainProgressPct", `${pct}%`);
    setText("explainProgressLabel", label);
  };

  const showStep = (i) => {
    if (stopped || i >= EXPLAIN_STEPS.length) return;
    const [pct, label, delay] = EXPLAIN_STEPS[i];
    setProgress(pct, label);
    if (delay) setTimeout(() => showStep(i + 1), delay);
  };

  showStep(0);
  return () => { stopped = true; setProgress(100, "Done!"); };
}

async function explainTerm() {
  const input = $("explainInput");
  const panel = $("explainContent");
  const paper = getActivePaper();
  if (!input || !panel) return;

  const term = input.value.trim();
  if (!term) { showToast("Please type a term to explain"); return; }
  if (!paper) { showToast("Upload or select a paper first"); return; }
  if (paper.indexStatus === "indexing") { showToast("Paper is still being indexed."); return; }

  const cacheKey = `${paper.id}::${term.toLowerCase()}`;
  if (state.explainCache[cacheKey]) { renderExplainResult(state.explainCache[cacheKey]); return; }

  clearElement(panel);
  panel.appendChild(makeExplainLoadingCard(term));
  const stopProgress = startExplainProgress();

  try {
    const data = await fetchJson("/api/explain", {
      method: "POST",
      body: JSON.stringify({ term, paper_path: paper.path }),
    });
    stopProgress();
    state.explainCache[cacheKey] = data;
    renderExplainResult(data);

    state.tabCache = {};  // server saved the term to Memory -> recalculate Insights/Prerequisites
    showToast(`"${term}" saved to Memory — Prerequisites and Difficulty Score will drop.`);
  } catch (error) {
    stopProgress();
    showBox("explainContent", "❌", "Explanation Failed", error.message || "Check your API key.");
  }
}

  if (cached !== null) { cfg.render(cached); return; }

  if (paper.indexStatus === "indexing") {
    showBox(cfg.panel, "⏳", "Indexing in progress...", `Wait for the green dot, then click ${cfg.label} again.`);
    return;
  }

  showBox(cfg.panel, "⏳", cfg.loadingTitle, cfg.loadingText);
  try {
    const data = await fetchJson(cfg.url, { method: "POST", body: JSON.stringify({ paper_path: paper.path }) });
    const result = cfg.parse(data, paper);
    setCachedTab(paper.id, name, result);
    cfg.render(result);
  } catch (error) {
    showBox(cfg.panel, "❌", cfg.failTitle, error.message || "Check your API key.");
  }
}

function switchTab(name) {
  TABS.forEach((tab) => {
    $(`tab-${tab}`)?.classList.toggle("active", tab === name);
    $(`panel-${tab}`)?.classList.toggle("active", tab === name);
  });

  if (name === "summary") showBox("summaryAccordion", "📝", "Summary — Coming Soon", "Auto-generated paper summaries are in development.", "In Development");
  if (name === "comparison") showBox("comparisonContent", "⚖️", "Comparison — Coming Soon", "Multi-paper comparison is in development.", "In Development");
  if (TAB_LOADERS[name]) void loadTab(name);
}

/* ── Loading + selecting papers ───────────────────────────── */

async function loadPapers() {
  const data = await fetchJson("/api/papers");
  state.papers = (Array.isArray(data.papers) ? data.papers : []).map((p) => ({ ...p, active: false }));

  const savedId = localStorage.getItem(STORAGE.activePaper);
  state.activePaperId = state.papers.some((p) => p.id === savedId) ? savedId : state.papers[0]?.id ?? null;
  state.papers = state.papers.map((p) => ({ ...p, active: p.id === state.activePaperId }));
  state.papers.filter((p) => p.indexStatus === "indexing").forEach(startIndexPolling);

  renderPapers();
  const paper = getActivePaper();
  setText("pdfTitle", paper ? paper.title : "Select a paper to view");
  await loadPdfInViewer(paper);
  await loadChatHistory(paper);
}

async function selectPaper(paperId) {
  setActivePaper(paperId);
  await loadChatHistory(getActivePaper());

  const activeTab = document.querySelector(".tab.active")?.id.replace("tab-", "");
  if (TAB_LOADERS[activeTab]) switchTab(activeTab);

  const paper = getActivePaper();
  if (paper?.indexStatus === "indexing") {
    showProgressBar(paper.indexPct ?? 0, paper.indexMessage || paper.indexStep || "Indexing...");
  } else {
    hideProgressBar();
  }
}

/* ── Upload ───────────────────────────────────────────────── */

async function handleFiles(files) {
  const pdfs = files.filter((f) => f.name.toLowerCase().endsWith(".pdf"));
  if (!pdfs.length) { showToast("Please upload PDF files only"); return; }

  for (const file of pdfs) {
    const formData = new FormData();
    formData.append("file", file);

    try {
      showToast(`Uploading ${file.name}...`);
      const response = await fetch(`${API_BASE}/api/upload`, { method: "POST", body: formData });
      const data = await response.json().catch(() => ({}));
      if (!response.ok) throw new Error(data.error || "Upload failed");

      const paper = { ...data.paper, analysis: {}, indexStatus: "indexing", indexPct: 5, active: true };
      state.papers = [paper, ...state.papers.map((p) => ({ ...p, active: false }))];
      state.activePaperId = paper.id;
      localStorage.setItem(STORAGE.activePaper, paper.id);

      renderPapers();
      setText("pdfTitle", paper.title);
      await loadPdfInViewer(paper);
      state.chatLoaded[paper.id] = true;
      state.chatHistory[paper.id] = [];
      renderChatHistory(paper.id);
      showProgressBar(5, "Starting...");
      showToast(`Uploaded ${file.name} — indexing in background...`);
      startIndexPolling(paper);
    } catch (error) {
      showToast(error.message || "Upload failed");
    }
  }
}

/* ── Chat ─────────────────────────────────────────────────── */

async function sendMessage() {
  if (state.thinking) return;

  const input = $("chatInput");
  const chatArea = $("chatArea");
  const thinking = $("thinkingMsg");
  const paper = getActivePaper();
  if (!input || !chatArea || !thinking) return;

  const message = input.value.trim();
  if (!message) return;
  if (!paper) { showToast("Upload or select a paper first"); return; }
  if (paper.indexStatus === "indexing") { showToast("Paper is still being indexed. Please wait."); return; }

  const addMessage = (role, text) => {
    (state.chatHistory[paper.id] ||= []).push({ role, text });
    if (state.activePaperId === paper.id) {
      chatArea.insertBefore(createMessageElement(role, text), thinking);
      chatArea.scrollTop = chatArea.scrollHeight;
    }
  };

  addMessage("user", message);
  input.value = "";
  input.style.height = "auto";
  thinking.style.display = "flex";
  chatArea.scrollTop = chatArea.scrollHeight;
  state.thinking = true;

  try {
    const data = await fetchJson("/api/chat", {
      method: "POST",
      body: JSON.stringify({ message, paper_path: paper.path }),
    });
    thinking.style.display = "none";
    addMessage("assistant", data.reply || "No response returned.");
  } catch (error) {
    thinking.style.display = "none";
    addMessage("assistant", error.message || "Error: Could not get response");
  } finally {
    state.thinking = false;
  }
}

function handleKey(event) {
  if (event.key === "Enter" && !event.shiftKey) { event.preventDefault(); void sendMessage(); }
}

}

function setupResizeHandle() {
  const handle = $("resizeHandle");
  const panel = $("pdfPanel");
  if (!handle || !panel) return;

  const setWidth = (px) => { panel.style.width = panel.style.minWidth = `${px}px`; };
  const saved = localStorage.getItem(STORAGE.pdfWidth);
  if (saved) setWidth(saved);

  let resizing = false, startX = 0, startWidth = 0;

  handle.addEventListener("mousedown", (e) => {
    resizing = true;
    startX = e.clientX;
    startWidth = panel.offsetWidth;
    document.body.style.cursor = "col-resize";
    document.body.style.userSelect = "none";
  });
  document.addEventListener("mousemove", (e) => {
    if (resizing) setWidth(Math.max(280, Math.min(700, startWidth + (startX - e.clientX))));
  });
  document.addEventListener("mouseup", () => {
    if (!resizing) return;
    resizing = false;
    document.body.style.cursor = document.body.style.userSelect = "";
    localStorage.setItem(STORAGE.pdfWidth, String(panel.offsetWidth));
  });
}

function setupUploadControls() {
  const fileInput = $("fileInput");
  const dropZone = $("dropZone");

  fileInput?.addEventListener("change", (e) => {
    void handleFiles(Array.from(e.target.files || []));
    e.target.value = "";
  });

  if (dropZone) {
    dropZone.addEventListener("dragover", (e) => { e.preventDefault(); dropZone.style.opacity = "0.7"; });
    dropZone.addEventListener("dragleave", () => { dropZone.style.opacity = "1"; });
    dropZone.addEventListener("drop", (e) => {
      e.preventDefault();
      dropZone.style.opacity = "1";
      void handleFiles(Array.from(e.dataTransfer.files || []));
    });
    dropZone.addEventListener("click", () => fileInput?.click());
  }

  $("pdfCloseBtn")?.addEventListener("click", () => {
    $("pdfPanel")?.style.setProperty("display", "none");
    $("resizeHandle")?.style.setProperty("display", "none");
    showToast("PDF panel hidden");
  });
}

/* ── Start ────────────────────────────────────────────────── */

document.addEventListener("DOMContentLoaded", async () => {

  try {
    await loadPapers();
  } catch (error) {
    showToast(error.message || "Failed to load papers");
    renderPapers();
  }
});

# pipeline/chunker.py

"""Split extracted paper text into overlapping chunks."""

from langchain_text_splitters import RecursiveCharacterTextSplitter

CHUNK_SIZE = 300
CHUNK_OVERLAP = 50
SEPARATORS = ["\n\n", "\n", " "]


def chunk_text(text: str) -> list[str]:
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=CHUNK_SIZE,
        chunk_overlap=CHUNK_OVERLAP,
        separators=SEPARATORS,
    )
    return splitter.split_text(text)

from werkzeug.security import check_password_hash, generate_password_hash

DB_PATH = Path(__file__).resolve().parent.parent / "users.db"

SCHEMA = """
CREATE TABLE IF NOT EXISTS users (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    username   TEXT NOT NULL UNIQUE COLLATE NOCASE,
    email      TEXT NOT NULL UNIQUE COLLATE NOCASE,
    password   TEXT NOT NULL,
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE TABLE IF NOT EXISTS known_terms (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id    INTEGER NOT NULL,
    term       TEXT NOT NULL COLLATE NOCASE,
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    UNIQUE(user_id, term),
    FOREIGN KEY(user_id) REFERENCES users(id)
);
CREATE TABLE IF NOT EXISTS user_settings (
    user_id        INTEGER PRIMARY KEY,
    gemini_api_key TEXT,
    FOREIGN KEY(user_id) REFERENCES users(id)
);
"""


@contextmanager
def db():
    """Open the database, save changes when done, always close it."""
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    try:
        with conn:  # commits on success, rolls back on error
            yield conn
    finally:
        conn.close()


def init_db() -> None:
    with db() as conn:
        conn.executescript(SCHEMA)

def _legacy_hash(password: str) -> str:
    """Old SHA-256 hash. Kept ONLY so old accounts can still log in once."""
    return hashlib.sha256(password.encode("utf-8")).hexdigest()


def create_user(username: str, email: str, password: str) -> dict:
    username, email = username.strip(), email.strip().lower()

    if len(username) < 3:
        return {"ok": False, "error": "Username must be at least 3 characters."}
    if len(password) < 6:
        return {"ok": False, "error": "Password must be at least 6 characters."}
    if "@" not in email:
        return {"ok": False, "error": "Please enter a valid email address."}

    try:
        with db() as conn:
            cur = conn.execute(
                "INSERT INTO users (username, email, password) VALUES (?, ?, ?)",
                (username, email, generate_password_hash(password)),
            )
        return {"ok": True, "user": {"id": cur.lastrowid, "username": username}}

    except sqlite3.IntegrityError as e:
        msg = str(e).lower()
        if "username" in msg:
            return {"ok": False, "error": "Username already taken. Please choose another."}
        if "email" in msg:
            return {"ok": False, "error": "An account with this email already exists."}
        return {"ok": False, "error": "Account creation failed. Please try again."}


def verify_user(username_or_email: str, password: str) -> dict:
    ident = username_or_email.strip()

    with db() as conn:
        row = conn.execute(
            "SELECT * FROM users WHERE username = ? COLLATE NOCASE OR email = ? COLLATE NOCASE",
            (ident, ident),
        ).fetchone()

        if not row:
            return {"ok": False, "error": "No account found with that username or email."}

        stored = row["password"]
        if hmac.compare_digest(stored, _legacy_hash(password)):
            # Old account: upgrade to the safer hash right now.
            conn.execute(
                "UPDATE users SET password = ? WHERE id = ?",
                (generate_password_hash(password), row["id"]),
            )
        elif not check_password_hash(stored, password):
            return {"ok": False, "error": "Incorrect password. Please try again."}

    return {"ok": True, "user": {"id": row["id"], "username": row["username"]}}

def add_known_term(user_id: int, term: str) -> None:
    term = term.strip().lower()
    if not term:
        return
    with db() as conn:
        conn.execute(
            "INSERT OR IGNORE INTO known_terms (user_id, term) VALUES (?, ?)",
            (user_id, term),
        )


def get_known_terms(user_id: int) -> list[str]:
    """All terms this user learned, newest first."""
    with db() as conn:
        rows = conn.execute(
            "SELECT term FROM known_terms WHERE user_id = ? ORDER BY created_at DESC",
            (user_id,),
        ).fetchall()
    return [r["term"] for r in rows]


def get_known_terms_set(user_id: int) -> set[str]:
    return set(get_known_terms(user_id))


def delete_known_term(user_id: int, term: str) -> bool:
    with db() as conn:
        cur = conn.execute(
            "DELETE FROM known_terms WHERE user_id = ? AND term = ?",
            (user_id, term.strip().lower()),
        )
        return cur.rowcount > 0


def delete_all_known_terms(user_id: int) -> int:
    with db() as conn:
        return conn.execute("DELETE FROM known_terms WHERE user_id = ?", (user_id,)).rowcount

    with db() as conn:
        cur = conn.execute(
            "UPDATE user_settings SET gemini_api_key = NULL WHERE user_id = ?", (user_id,)

        return cur.rowcount > 0

# pipeline/difficulty_scorer.py

"""Difficulty score = readability + uncommon words + technical terms + LLM opinion."""

import re

import nltk
from keybert import KeyBERT
from nltk.tokenize import sent_tokenize, word_tokenize
from wordfreq import word_frequency

from embedder import model as _EMBEDDING_MODEL  # reuse the shared model
from model_config import call_gemini
from personalization import difficulty_label

for pkg in ("punkt", "punkt_tab", "wordnet", "omw-1.4"):
    nltk.download(pkg, quiet=True)

KW_MODEL = KeyBERT(model=_EMBEDDING_MODEL)

WEIGHTS = {
    "readability": 0.20,
    "uncommon_words": 0.15,
    "technical_terms": 0.25,
    "llm_perception": 0.40,
}

UNCOMMON_FREQ_THRESHOLD = 0.000008
KEYBERT_TOP_N = 100
KEYBERT_TEXT_LIMIT = 8000
KEYBERT_MIN_SCORE = 0.3


def _clean_words(text: str) -> list[str]:
    """Lowercase alphabetic words longer than 2 letters."""
    return [w for w in word_tokenize(text.lower()) if w.isalpha() and len(w) > 2]


def _count_syllables(word: str) -> int:
    word = word.lower().strip(".,;:!?\"'")
    count, previous_was_vowel = 0, False
    for ch in word:
        is_vowel = ch in "aeiouy"
        if is_vowel and not previous_was_vowel:
            count += 1
        previous_was_vowel = is_vowel
    if word.endswith("e") and count > 1:
        count -= 1
    return max(1, count)


# ── 1. Readability ───────────────────────────────────────

def compute_readability_score(text: str) -> tuple[float, float, float]:
    """Returns (score 0-10, Flesch-Kincaid grade, Flesch reading ease)."""
    sentences = sent_tokenize(text)
    words = [w for w in word_tokenize(text) if w.isalpha()]
    if not sentences or not words:
        return 5.0, 5.0, 50.0

    words_per_sentence = len(words) / len(sentences)
    syllables_per_word = sum(_count_syllables(w) for w in words) / len(words)

    fk_grade = 0.39 * words_per_sentence + 11.8 * syllables_per_word - 15.59
    reading_ease = max(0.0, min(100.0, 206.835 - 1.015 * words_per_sentence - 84.6 * syllables_per_word))

    grade_part = min(10.0, max(0.0, fk_grade / 30.0 * 10.0))
    ease_part = min(10.0, max(0.0, (100.0 - reading_ease) / 14.0))
    return round(grade_part * 0.20 + ease_part * 0.80, 2), round(fk_grade, 2), round(reading_ease, 1)


# ── 2. Technical terms (KeyBERT) ─────────────────────────

def extract_technical_terms(text: str) -> list[str]:
    try:
        keywords = KW_MODEL.extract_keywords(
            text[:KEYBERT_TEXT_LIMIT],
            keyphrase_ngram_range=(1, 2),
            stop_words="english",
            top_n=KEYBERT_TOP_N,
            use_mmr=True,
            diversity=0.7,
        )
        return [kw for kw, score in keywords if score > KEYBERT_MIN_SCORE]
    except Exception as e:
        print(f"[KeyBERT] failed: {e}")
        return []


def compute_technical_term_score(words: list[str], terms: list[str]) -> float:
    if not words:
        return 5.0
    if not terms:
        return 0.0
    term_words = {w.lower() for term in terms for w in term.split() if len(w) > 3}
    density = sum(1 for w in words if w in term_words) / len(words)
    return round(min(10.0, density / 0.20 * 10.0), 2)


# ── 3. Uncommon words ────────────────────────────────────

def _is_uncommon(word: str) -> bool:
    return word_frequency(word, "en") < UNCOMMON_FREQ_THRESHOLD


def compute_uncommon_word_score(words: list[str], technical_words: set[str]) -> float:
    """Uncommon words, excluding technical jargon (already counted in technical terms)."""
    if not words:
        return 5.0
    uncommon = sum(1 for w in words if _is_uncommon(w) and w not in technical_words)
    return round(min(10.0, uncommon / len(words) * 20.0), 2)


# ── 4. LLM opinion ───────────────────────────────────────

def extract_opening_text(full_text: str) -> str:
    """Abstract + introduction (or the first 3000 characters)."""
    abstract = re.search(r"\bAbstract\b", full_text, re.IGNORECASE)
    intro = re.search(r"\b(1\.?\s*)?Introduction\b", full_text, re.IGNORECASE)
    next_section = re.compile(
        r"\b(2\.?\s*\w+|Related Work|Background|Methodology|Literature Review)\b", re.IGNORECASE
    )

    if abstract and intro:
        found = next_section.search(full_text, intro.end())
        end = found.start() if found else intro.end() + 2000
        return full_text[abstract.start():end].strip()
    if abstract:
        return full_text[abstract.start():abstract.start() + 2000].strip()
    if intro:
        found = next_section.search(full_text, intro.end())
        end = found.start() if found else intro.end() + 2000
        return full_text[intro.start():end].strip()
    return full_text[:3000].strip()


def compute_llm_score(opening_text: str, api_key: str) -> int | None:
    prompt = f"""You are a strict academic difficulty evaluator.
Rate the difficulty of this research paper opening from 1 to 10.

SCALE:
  1-2  = Beginner. Plain English, no background needed.
  3-4  = Undergraduate. Basic math/programming assumed.
  5-6  = Graduate. Assumes field foundations, dense with ideas.
  7-8  = Expert. Novel methods, heavy formalism, deep domain needed.
  9-10 = Highly specialized. Mastery of multiple subfields required.

RULES:
- Novel model/method/architecture proposed -> score at least 7.
- Applying existing methods to new data -> score 5-6.
- Heavy math/proofs -> add 1-2.
- Most published papers score 6-9.

Paper Opening:
{opening_text[:3000]}

Reply with ONLY a single integer from 1 to 10."""

    try:
        reply = call_gemini(prompt, api_key, temperature=0.0, tag="LLM")
    except Exception as e:
        print(f"[LLM] All models failed: {e}")
        return None

    match = re.search(r"\b(\d+)\b", reply)
    return max(1, min(10, int(match.group(1)))) if match else None


# ── Main ─────────────────────────────────────────────────

def analyze_difficulty(full_text: str, api_key: str) -> dict:
    words = _clean_words(full_text)

    readability, fk_grade, reading_ease = compute_readability_score(full_text)
    terms = extract_technical_terms(full_text)
    technical = compute_technical_term_score(words, terms)
    technical_words = {w.lower() for term in terms for w in term.split()}
    uncommon = compute_uncommon_word_score(words, technical_words)

    llm = compute_llm_score(extract_opening_text(full_text), api_key)
    if llm is None:
        llm = round((readability + uncommon + technical) / 3, 1)
        print(f"[Fallback] LLM unavailable, estimated llm_perception={llm}")

    scores = {
        "readability": readability,
        "uncommon_words": uncommon,
        "technical_terms": technical,
        "llm_perception": llm,
    }
    final = round(sum(scores[key] * WEIGHTS[key] for key in WEIGHTS), 2)

    uncommon_pct = round(sum(1 for w in words if _is_uncommon(w)) / max(len(words), 1) * 100, 1)

    return {
        "scores": scores,
        "weights": WEIGHTS,
        "final_score": final,
        "difficulty_label": difficulty_label(final),
        "breakdown": {
            "total_sentences": len(sent_tokenize(full_text)),
            "total_words": len(words),
            "uncommon_word_pct": uncommon_pct,
            "technical_keyphrases": len(terms),
            "technical_terms_list": terms,
            "flesch_kincaid_grade": fk_grade,
            "flesch_reading_ease": reading_ease,
        },
    }

# pipeline/embedder.py

"""Loads the embedding model + ChromaDB ONCE, and stores paper chunks."""

import os

os.environ["HF_HUB_OFFLINE"] = "1"        # don't touch the network
os.environ["TRANSFORMERS_OFFLINE"] = "1"

from pathlib import Path

import chromadb
from sentence_transformers import SentenceTransformer

EMBEDDING_MODEL_NAME = "BAAI/bge-base-en-v1.5"
CHROMA_PATH = Path(__file__).resolve().parent.parent / "chroma_db"

print("[embedder] Loading embedding model...")
model = SentenceTransformer(EMBEDDING_MODEL_NAME)
client = chromadb.PersistentClient(path=str(CHROMA_PATH))
print("[embedder] Model ready.")


def collection_exists_and_has_data(collection_name: str) -> bool:
    try:
        return client.get_collection(collection_name).count() > 0
    except Exception:
        return False


def embed_and_store(chunks: list[str], collection_name: str) -> None:
    """Replace the collection with fresh embeddings of `chunks`."""
    try:
        client.delete_collection(collection_name)
    except Exception:
        pass  # didn't exist yet

    collection = client.create_collection(collection_name)
    embeddings = model.encode(
        chunks, batch_size=64, show_progress_bar=False, convert_to_numpy=True
    ).tolist()

    collection.add(
        documents=chunks,
        embeddings=embeddings,
        ids=[f"chunk_{i}" for i in range(len(chunks))],
        metadatas=[{"chunk_index": i} for i in range(len(chunks))],
    )
    print(f"[embedder] Stored {len(chunks)} chunks in '{collection_name}'.")

# pipeline/extractor.py

"""Extract text from PDF files."""

import fitz
import pdfplumber

MIN_TEXT_LENGTH = 100


def extract_text_pymupdf(pdf_path: str) -> str:
    text_parts = []
    with fitz.open(pdf_path) as document:
        for page in document:
            text_parts.append(page.get_text())
    return "".join(text_parts)


def extract_text_pdfplumber(pdf_path: str) -> str:
    text_parts = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            page_text = page.extract_text()
            if page_text:
                text_parts.append(page_text + "\n")
    return "".join(text_parts)


def extract_text(pdf_path: str) -> str:
    text = extract_text_pymupdf(pdf_path).strip()
    if len(text) < MIN_TEXT_LENGTH:
        text = extract_text_pdfplumber(pdf_path).strip()
    return text

# pipeline/model_config.py

"""Shared Gemini settings + ONE helper that every module uses to call Gemini."""

import time

from google import genai
from google.genai import types

GEMINI_MODEL_POOL = [
    "gemini-3.1-flash-lite",
    "gemini-2.5-flash",
    "gemini-3-flash",
    "gemini-2.5-flash-lite",
]

MAX_RETRIES_PER_MODEL = 3
RETRY_DELAY_SECONDS = 1

# Temporary problems: wait and retry the same model.
_TEMPORARY_ERRORS = ("429", "quota", "503", "unavailable")


def call_gemini(prompt: str, api_key: str, *, temperature: float | None = None,
                max_output_tokens: int | None = None, tag: str = "Gemini") -> str:
    """Send a prompt to Gemini. Retries on temporary errors, then tries the next model."""
    if not api_key:
        raise ValueError("No Gemini API key configured. Add one via the gear icon or .env")

    client = genai.Client(api_key=api_key)
    config = types.GenerateContentConfig(
        temperature=temperature,
        max_output_tokens=max_output_tokens,
    )
    last_error = None

    for model in GEMINI_MODEL_POOL:
        for attempt in range(1, MAX_RETRIES_PER_MODEL + 1):
            try:
                response = client.models.generate_content(
                    model=model, contents=prompt, config=config
                )
                if response.text:
                    return response.text.strip()
                break  # empty answer -> try the next model

            except Exception as error:
                last_error = error
                print(f"[{tag}] {model} (attempt {attempt}/{MAX_RETRIES_PER_MODEL}): {str(error)[:200]}")
                if not any(word in str(error).lower() for word in _TEMPORARY_ERRORS):
                    break  # not temporary -> next model
                time.sleep(RETRY_DELAY_SECONDS)

    raise RuntimeError(f"All Gemini models failed. Last error: {last_error}")

# pipeline/personalization.py

"""Personalization: what the user already knows lowers difficulty + hides prerequisites."""

import copy
import re

DIFFICULTY_LABELS = [(3.5, "Easy"), (6.5, "Medium"), (10.0, "Hard")]

# Knowing ALL prerequisites can lower the LLM score by at most this much (50%).
MAX_LLM_REDUCTION = 0.5

# Words that don't change which concept a phrase is about.
GENERIC_WORDS = {
    "mechanism", "function", "architecture", "model", "network", "layer",
    "method", "technique", "concept", "basic", "algorithm", "process",
    "and", "of", "the", "for", "in", "to", "a", "on", "with",
}

PREREQ_LINE = re.compile(r"^\s*\d+[.)]\s*(.+)")


def difficulty_label(score: float) -> str:
    for threshold, label in DIFFICULTY_LABELS:
        if score <= threshold:
            return label
    return "Very Hard"


def _core_tokens(text: str) -> set[str]:
    """Lowercase words, plural -> singular, generic words removed."""
    tokens = set()
    for word in re.findall(r"[a-z0-9]+", (text or "").lower()):
        if len(word) > 3 and word.endswith("s") and not word.endswith("ss"):
            word = word[:-1]
        if word not in GENERIC_WORDS:
            tokens.add(word)
    return tokens


def terms_match(known: str, candidate: str, loose: bool = False) -> bool:
    """
    True if the user's known term means the same concept as `candidate`.
      strict (prerequisites): same core words, or the smaller phrase has 2+ words
                              and sits fully inside the bigger one.
      loose  (KeyBERT terms): any containment counts.
    """
    k, c = _core_tokens(known), _core_tokens(candidate)
    if not k or not c:
        return False
    if k == c:
        return True
    small, big = (k, c) if len(k) <= len(c) else (c, k)
    return small <= big and (loose or len(small) >= 2)


def prereq_topic(line_text: str) -> str:
    """'Softmax Function: Mathematical tool...' -> 'Softmax Function'"""
    return line_text.split(":")[0].strip()


def parse_prereq_topics(prereq_text: str | None) -> list[str]:
    topics = []
    for line in (prereq_text or "").split("\n"):
        match = PREREQ_LINE.match(line)
        if match:
            topics.append(prereq_topic(match.group(1)))
    return topics


def personalize_analysis(base_result: dict, known_terms: set[str],
                         base_prereq_text: str | None = None) -> dict:
    """Returns a COPY of the base analysis, lowered for what this user knows."""
    result = copy.deepcopy(base_result)
    if not known_terms:
        return result

    scores, weights = result["scores"], result["weights"]

    tech_terms = result.get("breakdown", {}).get("technical_terms_list") or []
    tech_matched = sum(
        1 for term in tech_terms if any(terms_match(k, term, loose=True) for k in known_terms)
    )

    prereqs = parse_prereq_topics(base_prereq_text)
    prereq_matched = sum(1 for t in prereqs if any(terms_match(k, t) for k in known_terms))

    if not tech_matched and not prereq_matched:
        return result

    if tech_matched:
        scores["technical_terms"] = round(
            scores["technical_terms"] * (1 - tech_matched / len(tech_terms)), 2
        )
    if prereq_matched:
        cut = MAX_LLM_REDUCTION * prereq_matched / len(prereqs)
        scores["llm_perception"] = round(scores["llm_perception"] * (1 - cut), 2)

    final = round(sum(scores[key] * weights[key] for key in weights), 2)

    result["final_score"] = final
    result["difficulty_label"] = difficulty_label(final)
    result["personalization"] = {
        "known_terms_matched": tech_matched,
        "total_technical_terms": len(tech_terms),
        "prereqs_known": prereq_matched,
        "total_prereqs": len(prereqs),
        "base_final_score": base_result["final_score"],
    }
    return result


def personalize_prerequisites(base_text: str, known_terms: set[str]) -> str:
    """Remove prerequisites the user already knows, then renumber the list."""
    if not base_text or not known_terms:
        return base_text

    lines, number = [], 0
    for line in base_text.strip().split("\n"):
        match = PREREQ_LINE.match(line)
        if not match:
            lines.append(line)
            continue
        if any(terms_match(k, prereq_topic(match.group(1))) for k in known_terms):
            continue
        number += 1
        lines.append(f"{number}. {match.group(1)}")
    return "\n".join(lines)

# pipeline/prerequisite_extractor.py

"""Prerequisite Extractor: turns a paper into a beginner -> advanced learning roadmap."""

import re

from model_config import call_gemini

STOP_WORDS = ("related work", "methodology", "methods", "approach", "experiments", "results")


def clean_text(text: str) -> str:
    text = re.sub(r"\s+", " ", text)
    text = re.sub(r"([a-z])([A-Z])", r"\1 \2", text)
    text = re.sub(r"\[\d+\]", "", text)
    text = re.sub(r"http\S+", "", text)
    return text.strip()


def extract_abstract(text: str) -> str:
    start = text.lower().find("abstract")
    if start == -1:
        return text[:4000]
    chunk = text[start:start + 6000]
    end = chunk.lower().find("introduction")
    return (chunk[:end] if end != -1 else chunk).strip()


def extract_introduction(text: str) -> str:
    start = text.lower().find("introduction")
    if start == -1:
        return text[:6000]
    chunk = text[start:start + 10000]
    lower = chunk.lower()
    positions = [lower.find(w) for w in STOP_WORDS if w in lower]
    return (chunk[:min(positions)] if positions else chunk).strip()


def build_concept_prompt(abstract: str, intro: str) -> str:
    return f"""
Extract the core concepts required to understand this research paper.

RULES:
- Return 15 to 25 concepts, one per line.
- No explanations, numbering, bullet points, or markdown.
- Focus on technical concepts, theories, algorithms, methods, and prerequisite knowledge.

ABSTRACT:
{abstract}

INTRODUCTION:
{intro}
"""


def build_prerequisite_prompt(concepts: str) -> str:
    return f"""
You are an academic prerequisite extraction system.
Convert the concepts below into a learning roadmap.

RULES:
- Output exactly 12 to 15 prerequisites, ordered from beginner to advanced.
- Format:

1. Topic: Short explanation

- One line per prerequisite. Keep explanations short.
- No markdown, bullet points, headings, or extra text. Output ONLY the numbered list.

CONCEPTS:
{concepts}
"""


def clean_output(text: str) -> str:
    """Remove markdown/duplicates and renumber the list 1, 2, 3..."""
    cleaned, seen = [], set()
    for line in text.split("\n"):
        line = re.sub(r"\*\*(.*?)\*\*", r"\1", line.strip())
        line = re.sub(r"^[-*]\s*", "", line)
        line = re.sub(r"^\d+[.)]\s*", "", line)
        if line and line.lower() not in seen:
            seen.add(line.lower())
            cleaned.append(f"{len(cleaned) + 1}. {line}")
    return "\n".join(cleaned)


def extract_prerequisites(text: str, api_key: str) -> str:
    text = clean_text(text)
    concepts = call_gemini(
        build_concept_prompt(extract_abstract(text), extract_introduction(text)),
        api_key, tag="Prereq",
    )
    roadmap = call_gemini(build_prerequisite_prompt(concepts), api_key, tag="Prereq")
    return clean_output(roadmap)

"""Answer questions about a paper using Gemini, strictly from the retrieved chunks."""

def build_prompt(question: str, context_chunks: list[str]) -> str:
    context = "\n\n---\n\n".join(
        f"[Chunk {i}]\n{chunk.strip()}" for i, chunk in enumerate(context_chunks, start=1)
    )

    return f"""You are an expert AI assistant specialized in answering questions about research papers.
Your answer must be grounded exclusively in the context below. You have no outside knowledge.

=== RESEARCH PAPER CONTEXT ===
{context}

=== USER QUESTION ===
{question}

=== HOW TO ANSWER (internal reasoning, never show these steps) ===
Decide the question type, find the relevant chunks, combine them, then answer:
- Factual: answer directly with the exact value, name, or finding. Nothing extra.
- Conceptual: explain using only the paper's own definitions. No outside analogies.
- Comparative: describe each side separately, then state the contrast.
- Evaluative: report only what the authors explicitly state. No own judgment.

=== HALLUCINATION CONTROL (STRICT) ===
- NEVER use outside knowledge. NEVER assume or invent details.
- Context fully answers the question: give a complete answer.
- Context partly answers it: answer with what exists, then add exactly:
  "Note: The provided context does not contain complete information about [the missing aspect]."
- Topic appears in the context but has no relevant info: respond exactly:
  "The provided document does not contain information relevant to this question."
- Topic does not appear in the context at all: respond exactly:
  "This question is unrelated to the research paper provided."

=== OUTPUT FORMAT ===
- Clear, professional, academic English. Start directly with the answer
  (never "Based on the context" or "According to the document").
- Light markdown: **bold** for key terms and numbers, numbered lists for steps,
  bullets for unordered items, short paragraphs. Use a heading (## ) only for 2+ sub-topics.
- A one-line factual answer should be plain text. Do not repeat yourself.
- Short for factual questions, thorough for conceptual or evaluative ones.

=== ANSWER ==="""


def answer_question(question: str, context_chunks: list[str], api_key: str) -> str:
    if not question or not question.strip():
        raise ValueError("Question cannot be empty.")
    if not context_chunks:

    prompt = build_prompt(question, context_chunks)

# pipeline/retriever.py

"""Find the most relevant chunks of a paper in ChromaDB."""

import re

from embedder import client, model  # shared model: no second copy loaded

SUMMARY_PATTERNS = (
    "what is this paper about", "what does this paper", "what is the paper about",
    "summarize", "summarise", "summary", "overview", "outline", "brief", "tldr", "tl;dr",
    "what does this document", "what is this document", "tell me about this paper",
    "explain this paper", "what is the main", "what are the main",
    "what is the purpose", "what is the goal",
)

LOW_SIGNAL_PATTERNS = (
    r"\bprovided proper attribution\b", r"\breferences\b", r"\bbibliography\b",
    r"\bconference on\b", r"\bproceedings\b", r"\barxiv\b", r"\bdoi\b",
    r"\bfigure\s+\d+", r"\bfig\.\s*\d+", r"\btable\s+\d+", r"\bwork performed while\b",
)

HIGH_SIGNAL_PATTERNS = (
    r"\babstract\b", r"\bintroduction\b", r"\bconclusion\b", r"\bwe propose\b",
    r"\bwe present\b", r"\bwe introduce\b", r"\bthis paper\b",
    r"\bour contributions\b", r"\bexperiments\b", r"\bresults\b",
)

QUERY_PREFIX = "Represent this sentence for searching relevant passages: "


def is_summary_question(query: str) -> bool:
    q = query.lower().strip()
    return any(p in q for p in SUMMARY_PATTERNS)


def score_low_signal(chunk: str) -> int:
    """Higher = more like references/boilerplate."""
    text = chunk.lower()
    score = sum(1 for p in LOW_SIGNAL_PATTERNS if re.search(p, text))

    if len(re.findall(r"\b(?:19|20)\d{2}\b", text)) >= 5:
        score += 2
    if sum(c.isdigit() for c in text) / max(len(text), 1) > 0.12:
        score += 2

    words = re.findall(r"[a-zA-Z]+", text)
    if words and len(set(words)) / len(words) < 0.35:
        score += 1
    return score


def score_high_signal(chunk: str) -> int:
    text = chunk.lower()
    return sum(1 for p in HIGH_SIGNAL_PATTERNS if re.search(p, text))


def lexical_overlap_score(query: str, chunk: str) -> int:
    q = set(re.findall(r"[a-zA-Z]+", query.lower()))
    c = set(re.findall(r"[a-zA-Z]+", chunk.lower()))
    return len(q & c)


def get_opening_chunks(collection, count: int = 3) -> list[str]:
    try:
        result = collection.get(ids=[f"chunk_{i}" for i in range(count)])
    except Exception:
        return []
    pairs = sorted(zip(result.get("ids", []), result.get("documents", [])),
                   key=lambda p: int(p[0].split("_")[-1]))
    return [doc for _, doc in pairs if doc]


def dedupe_chunks(chunks: list[str]) -> list[str]:
    seen, unique = set(), []
    for chunk in chunks:
        key = re.sub(r"\s+", " ", chunk).strip()[:500]
        if key and key not in seen:
            seen.add(key)
            unique.append(chunk)
    return unique


def retrieve_relevant_chunks(query: str, collection_name: str, top_k: int = 10) -> list[str]:
    collection = client.get_or_create_collection(collection_name)
    summary_mode = is_summary_question(query)

    search_query = query
    if summary_mode:
        search_query += (". abstract introduction objective methodology approach "
                         "contributions experiments results conclusion paper summary")

    embedding = model.encode([QUERY_PREFIX + search_query]).tolist()
    results = collection.query(
        query_embeddings=embedding,
        n_results=max(top_k * 5, 30 if summary_mode else 20),
    )

    ranked = []
    for rank, (chunk, distance) in enumerate(zip(results["documents"][0], results["distances"][0])):
        score = distance - lexical_overlap_score(query, chunk) * 0.1
        if summary_mode:
            score += score_low_signal(chunk) * 0.35 - score_high_signal(chunk) * 0.2
        ranked.append((score, rank, chunk))
    ranked.sort(key=lambda item: (item[0], item[1]))

    selected = [c for _, _, c in ranked if not summary_mode or score_low_signal(c) < 2]
    if summary_mode:
        selected = get_opening_chunks(collection) + selected

    return dedupe_chunks(selected)[:max(top_k, 10)]

# pipeline/term_explainer.py

"""Term Explainer: what the paper says + simple words + a real-world example."""

import re

from model_config import call_gemini

SECTIONS = ("FROM_PAPER", "SIMPLE_EXPLANATION", "REAL_WORLD_EXAMPLE")


def build_explain_prompt(term: str, chunks: list[str], paper_title: str = "") -> str:
    if chunks:
        context = "\n\n---\n\n".join(
            f"[Chunk {i}]\n{chunk.strip()}" for i, chunk in enumerate(chunks, start=1)
        )
    else:
        context = "No relevant chunks were found in this paper for the given term."

    paper_hint = f'The paper is titled: "{paper_title}".' if paper_title else ""
    domain = f"the paper domain ({paper_title})" if paper_title else "the research paper domain"

    return f"""You are an expert academic tutor helping a university student understand research papers.
The student wants to understand the term: "{term}"
{paper_hint}

=== PAPER CONTEXT (retrieved chunks) ===
{context}

=== YOUR TASK ===
Respond in EXACTLY this format. Do not change the tags or add sections:

FROM_PAPER_START
[What this paper says about "{term}". Be honest:
- Chunks explain it clearly: summarize in 2-4 sentences.
- Chunks only mention it: say "This paper mentions '{term}' but does not explain it in detail."
- No relevant chunks: say "This term does not appear to be discussed in this paper."]
FROM_PAPER_END

SIMPLE_EXPLANATION_START
[A plain English explanation of "{term}" in 3-5 sentences.
- Everyday language a high school student understands. No jargon unless explained.
- Frame it in {domain}.
- Do NOT include an example here.]
SIMPLE_EXPLANATION_END

REAL_WORLD_EXAMPLE_START
[ONE concrete real-world example or analogy. Start with "Example:".
- Vivid, specific, relatable (everyday life, cooking, sports, travel...).
- 3-5 sentences. No jargon at all.]
REAL_WORLD_EXAMPLE_END

=== STRICT RULES ===
- Never invent what the paper says. Only report what is in the chunks.
- The simple explanation and example must be written even if the paper does not cover the term.
- No markdown (no **, no #, no * bullets). No text outside the three tagged sections.
- Write in clear, warm, encouraging academic English.
"""


def parse_explain_response(raw: str) -> dict:
    """Pull the three tagged sections out of Gemini's reply."""
    parts = {}
    for name in SECTIONS:
        match = re.search(rf"{name}_START\s*(.*?)\s*{name}_END", raw, re.DOTALL)
        parts[name] = match.group(1).strip() if match else ""

    if not all(parts.values()):  # tags missing -> split the text in 3 as a fallback
        lines = [line.strip() for line in raw.strip().split("\n") if line.strip()]
        third = len(lines) // 3
        fallback = (
            " ".join(lines[:third]) or raw.strip(),
            " ".join(lines[third:2 * third]) or raw.strip(),
            " ".join(lines[2 * third:]),
        )
        for name, text in zip(SECTIONS, fallback):
            parts[name] = parts[name] or text

    return {
        "from_paper": parts["FROM_PAPER"],
        "simple_explanation": parts["SIMPLE_EXPLANATION"],
        "real_world_example": parts["REAL_WORLD_EXAMPLE"],
    }


def explain_term(term: str, chunks: list[str], api_key: str, paper_title: str = "") -> dict:
    term = (term or "").strip()
    if not term:
        raise ValueError("Term cannot be empty.")

    raw = call_gemini(build_explain_prompt(term, chunks, paper_title), api_key, tag="TermExplainer")

    return {
        "term": term,
        **parse_explain_response(raw),
        "chunks_used": len(chunks),
        "found_in_paper": bool(chunks),
    }

# README.md

# Paper Insight AI

An AI-powered research paper assistant built with Flask, RAG (Retrieval-Augmented Generation), and Google Gemini. Upload any research paper PDF and get an interactive workspace to chat with it, analyze its difficulty, extract prerequisite knowledge, and get plain-English explanations of any technical term — all grounded strictly in the paper's own content.

## Features

- **Chat / Q&A** — Ask questions about the paper and get grounded, hallucination-resistant answers using Gemini + RAG retrieval over the paper's content.
- **Difficulty Scorer** — A weighted difficulty score (1–10) combining readability (Flesch-Kincaid), uncommon word density, technical term density (KeyBERT), and an LLM-based difficulty judgment.
- **Prerequisites Extractor** — Generates a 12–15 topic learning roadmap (beginner → advanced) needed to understand the paper.
- **Term Explainer** — Explains any term in three layers: what the paper says about it, a simple plain-English explanation, and a real-world analogy.
- **PDF Viewer** — Built-in zoomable PDF viewer with fullscreen mode, powered by PDF.js.
- **Authentication** — Per-user accounts with isolated paper storage (SQLite + hashed passwords).
- **Persistent Caching** — Analysis, prerequisites, and chat history are cached to disk so results survive server restarts and aren't recomputed.

## Tech Stack

| Layer | Technology |
|---|---|
| Backend | Flask (Python) |
| Embeddings | `BAAI/bge-base-en-v1.5` (Sentence-Transformers) |
| Vector Store | ChromaDB |
| Text Splitting | LangChain `RecursiveCharacterTextSplitter` (chunk size 300, overlap 50) |
| LLM | Google Gemini (model pool with automatic fallback) |
| Keyword Extraction | KeyBERT |
| PDF Extraction | PyMuPDF (primary), pdfplumber (fallback) |
| Auth / Storage | SQLite |
| Frontend | Vanilla HTML/CSS/JS, PDF.js |
